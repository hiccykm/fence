貼入 Apps Script 並部署為「網頁應用程式」（誰可以存取：所有人）。金鑰請填在「專案設定 → 指令碼屬性」：

/**
 * AI 電子圍籬 Webhook：Google Sheet + Drive + ImgBB + LINE（診斷版）
 *
 * 指令碼屬性（專案設定 → 指令碼屬性）：
 *   LINE_CHANNEL_ACCESS_TOKEN / LINE_TARGET_ID
 *   LINE_CHANNEL_ACCESS_TOKEN2 / LINE_TARGET_ID2
 *   FOLDER_ID / IMGBB_API_KEY
 *
 * 網頁傳入 lineTarget：ALL 全部 / NONE 全不要 / LINE1 第一組 / LINE2 第二組
 *
 * 每一筆紀錄的 H 欄「Status」會寫出各步驟成功或失敗的原因，
 * 網頁端也會顯示同樣的回應，方便排查。

var SPREADSHEET_ID = '19vGOlRS7xNfncFF3Nt3XYcYKHjVdbAHIaba1RWG1gVw';
var SHEET_NAME = 'Intrusion Log';
var HEADERS = ['日期時間', '事件類型', '區域', '目標', '來源', 'Drive URL', 'ImgBB URL', 'Status'];

function doGet() {
  return json_({ ok: true, message: 'AI Geofence API is running' });
}

function doPost(e) {
  var status = [];
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return json_({ ok: false, error: 'No POST data' });
    }
    var data = JSON.parse(e.postData.contents);
    var event = normalizeEvent_(data);
    var photoLen = data.photo ? String(data.photo).length : 0;
    status.push('photo=' + (photoLen ? Math.round(photoLen / 1024) + 'KB' : 'none'));

    var driveUrl = '', imgbbUrl = '';

    // lineTarget：ALL（全部）/ NONE（全不要）/ LINE1（第1組）/ LINE2（第2組）
    var lineOff = (event.lineTarget === 'NONE');

    // 1) 先立即推送 LINE 文字，不等圖片上傳
    if (lineOff) {
      status.push('line=skipped(全不要)');
    } else {
      try {
        var lineTextResult = sendLineText_(event);
        status.push('lineText=ok(' + lineTextResult.channels.join(',') + ')');
        if (lineTextResult.failures.length) status.push('lineTextWarning=' + lineTextResult.failures.join(' | '));
      } catch (err) {
        status.push('lineText=FAIL ' + String(err).substring(0, 300));
      }
    }

    if (photoLen) {
      // 2) 上傳圖片到 ImgBB，成功後再補推 LINE 圖片
      var im = uploadToImgbb_(data.photo);
      imgbbUrl = im.url;
      status.push('imgbb=' + (im.url ? 'ok' : 'FAIL ' + im.error));
      if (im.url && !lineOff) {
        try {
          var lineImageResult = sendLineImage_(im.url, event.lineTarget);
          status.push('lineImage=ok(' + lineImageResult.channels.join(',') + ')');
          if (lineImageResult.failures.length) status.push('lineImageWarning=' + lineImageResult.failures.join(' | '));
        } catch (err2) {
          status.push('lineImage=FAIL ' + String(err2).substring(0, 300));
        }
      }
      // 3) 備份到 Google Drive
      var d = saveToDrive_(data.photo, event.timestamp);
      driveUrl = d.url;
      status.push('drive=' + (d.url ? 'ok' : 'FAIL ' + d.error));
    }

    getLogSheet_().appendRow([
      event.timestamp, event.eventType, event.zone, event.target,
      event.source, driveUrl, imgbbUrl, status.join(' | ')
    ]);

    return json_({ ok: true, status: status, driveUrl: driveUrl, imgbbUrl: imgbbUrl });
  } catch (error) {
    console.error(error);
    try {
      getLogSheet_().appendRow([nowTaipei_(), 'ERROR', '', '', '', '', '', String(error)]);
    } catch (e2) {}
    return json_({ ok: false, error: String(error), status: status });
  }
}

function normalizeEvent_(data) {
  return {
    timestamp: data.timestamp || data.time || nowTaipei_(),
    eventType: data.eventType || data.event || data.type || 'Geofence intrusion',
    zone: data.zone || 'Zone B',
    target: data.target || 'Unknown target',
    source: data.source || 'AI geofence webpage',
    label: data.label || '',
    lineTarget: normalizeLineTarget_(data.lineTarget)
  };
}

function normalizeLineTarget_(v) {
  var t = String(v || 'ALL').toUpperCase();
  return (t === 'ALL' || t === 'NONE' || t === 'LINE1' || t === 'LINE2') ? t : 'ALL';
}

function getLogSheet_() {
  var book = SpreadsheetApp.openById(SPREADSHEET_ID);
  var sheet = book.getSheetByName(SHEET_NAME) || book.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(HEADERS);
  } else if (!sheet.getRange(1, 8).getValue()) {
    sheet.getRange(1, 8).setValue('Status');
  }
  return sheet;
}

// 回傳 {url, error}
function saveToDrive_(base64Image, timestamp) {
  var folderId = PropertiesService.getScriptProperties().getProperty('FOLDER_ID');
  if (!folderId) return { url: '', error: 'FOLDER_ID missing' };
  try {
    var bytes = Utilities.base64Decode(String(base64Image).split(',').pop());
    var folder = DriveApp.getFolderById(folderId);
    var name = 'geofence_' + String(timestamp).replace(/[^0-9]/g, '') + '.jpg';
    var file = folder.createFile(Utilities.newBlob(bytes, 'image/jpeg', name));
    return { url: file.getUrl(), error: '' };
  } catch (error) {
    console.error('Drive error: ' + error);
    return { url: '', error: String(error).substring(0, 150) };
  }
}

// 回傳 {url, error}
function uploadToImgbb_(base64Image) {
  var key = PropertiesService.getScriptProperties().getProperty('IMGBB_API_KEY');
  if (!key) return { url: '', error: 'IMGBB_API_KEY missing' };
  var cleanBase64 = String(base64Image).split(',').pop();
  var lastError = '';
  // 黃色警報通常由自動偵測觸發，若 ImgBB 短暫回應失敗，重試一次避免只有文字沒有圖片。
  for (var attempt = 1; attempt <= 2; attempt++) {
    try {
      var response = UrlFetchApp.fetch('https://api.imgbb.com/1/upload?key=' + encodeURIComponent(key), {
        method: 'post',
        contentType: 'application/x-www-form-urlencoded',
        payload: { image: cleanBase64 },
        muteHttpExceptions: true
      });
      var code = response.getResponseCode();
      var body = response.getContentText();
      console.log('ImgBB response attempt ' + attempt + ' ' + code + ': ' + body);
      var result = JSON.parse(body);
      if (result.success && result.data) {
        // LINE 必須收到可直接讀取的 HTTPS 圖片網址；url 是 ImgBB 的原始圖片 URL。
        var directUrl = result.data.url || result.data.display_url || '';
        if (/^https:\/\//i.test(directUrl)) return { url: directUrl, error: '' };
        lastError = 'ImgBB returned no HTTPS image URL';
      } else {
        lastError = 'HTTP ' + code + ' ' + body.substring(0, 150);
      }
    } catch (error) {
      lastError = String(error).substring(0, 150);
      console.error('ImgBB attempt ' + attempt + ' error: ' + error);
    }
    if (attempt < 2) Utilities.sleep(500);
  }
  return { url: '', error: lastError || 'ImgBB upload failed' };
}

function linePush_(messages, lineTarget) {
  var props = PropertiesService.getScriptProperties();
  var channels = [
    {
      name: 'LINE1',
      token: String(props.getProperty('LINE_CHANNEL_ACCESS_TOKEN') || '').trim(),
      targetId: String(props.getProperty('LINE_TARGET_ID') || '').trim()
    },
    {
      name: 'LINE2',
      token: String(props.getProperty('LINE_CHANNEL_ACCESS_TOKEN2') || '').trim(),
      targetId: String(props.getProperty('LINE_TARGET_ID2') || '').trim()
    }
  ];

  var sent = [];
  var failures = [];
  var target = String(lineTarget || 'ALL').toUpperCase();
  channels.forEach(function(channel) {
    if (target !== 'ALL' && target !== channel.name) return;
    // 第二組尚未設定時略過，讓第一組仍可正常運作。
    if (!channel.token && !channel.targetId) return;
    if (!channel.token || !channel.targetId) {
      failures.push(channel.name + ' token/target 未完整設定');
      return;
    }

    try {
      var response = UrlFetchApp.fetch('https://api.line.me/v2/bot/message/push', {
        method: 'post',
        contentType: 'application/json',
        headers: { Authorization: 'Bearer ' + channel.token },
        payload: JSON.stringify({ to: channel.targetId, messages: messages }),
        muteHttpExceptions: true
      });
      var code = response.getResponseCode();
      var body = response.getContentText();
      console.log(channel.name + ' response ' + code + ': ' + body);
      if (code < 200 || code >= 300) {
        failures.push(channel.name + ' LINE API ' + code + '（' + lineHint_(code) + '）: ' + body.substring(0, 160));
      } else {
        sent.push(channel.name);
      }
    } catch (error) {
      failures.push(channel.name + ': ' + String(error).substring(0, 180));
    }
  });

  if (!sent.length) {
    throw new Error('沒有任何 LINE 通知成功：' + failures.join(' | '));
  }
  // 部分成功時仍回傳成功群組，讓第一組不會因第二組 token 錯誤而被誤判為全部失敗。
  return { sent: true, channels: sent, failures: failures };
}

function lineHint_(code) {
  if (code === 400) return 'Target ID 不屬於這個 Bot：請用該 Bot 所屬 Provider 取得的 userId，且對方須已加入此官方帳號為好友';
  if (code === 401) return 'Channel Access Token 無效、過期或貼錯';
  if (code === 403) return '對方未加此官方帳號好友或已封鎖，或沒有推播權限';
  if (code === 429) return '本月推播額度已用完或太頻繁';
  return '請對照 LINE Messaging API 錯誤碼';
}

function sendLineText_(event) {
  var text = '🚨 校園電子圍籬警報\n' +
    (event.label ? '事件：' + event.label + '\n' : '事件：' + event.eventType + '\n') +
    '時間：' + event.timestamp + '\n' +
    '區域：' + event.zone + '\n' +
    '目標：' + event.target + '\n' +
    '來源：' + event.source;
  return linePush_([{ type: 'text', text: text }], event.lineTarget);
}

function sendLineImage_(imageUrl, lineTarget) {
  return linePush_([{ type: 'image', originalContentUrl: imageUrl, previewImageUrl: imageUrl }], lineTarget);
}

function nowTaipei_() {
  return Utilities.formatDate(new Date(), 'Asia/Taipei', 'yyyy-MM-dd HH:mm:ss');
}

function json_(value) {
  return ContentService.createTextOutput(JSON.stringify(value))
    .setMimeType(ContentService.MimeType.JSON);
}
